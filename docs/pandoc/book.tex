% ---- the book: typesetting on top of header.tex (xelatex, book class) -----
% Passed after header.tex by scripts/book/pdf.py, so these settings win where
% the two overlap. The book is a textbook, not a manual: running heads carry
% the chapter, chapters open on a right-hand page, figures are centred and a
% little smaller than the text block.

\setkeys{Gin}{width=0.9\linewidth,height=0.45\textheight,keepaspectratio}

\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[LE]{\small\nouppercase{\leftmark}}
\fancyhead[RO]{\small\nouppercase{\rightmark}}
\fancyfoot[LE,RO]{\small\thepage}
\fancyfoot[RE,LO]{\footnotesize From First Principles to Intelligent Machines}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[LE,RO]{\small\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
}

% Block quotes are used for definitions and key results: set them off.
\usepackage{xcolor}
\definecolor{bookrule}{HTML}{2B6CB0}
\renewenvironment{quote}%
  {\par\medskip\noindent\begin{minipage}{\linewidth}\leftskip=1.2em\rightskip=1em%
   \color{black}\small}%
  {\end{minipage}\par\medskip}
