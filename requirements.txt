# Python tooling for brain's benchmarks, parity checks, and the NPU runtime.
#
# The Rust engine itself needs NONE of these — `make release` builds it with
# cargo alone. These power the tools/ scripts (inference benchmarks, parity
# dumps) and provide the OpenVINO runtime that `brain … --device npu` loads at
# run time (the pip `openvino` wheel bundles the native libraries the Rust
# `openvino` crate dlopens). Install with `make requirements`.
#
# Kept in sync with what tools/**/*.py and scripts/**/*.py actually import —
# `huggingface_hub` was removed from here because nothing imports it (brain
# fetches checkpoints itself, via `brain fetch` / scripts/fetch-testdata.sh).

openvino>=2024.0       # NPU/OpenVINO runtime (native libs for `--device npu`)
torch                  # HuggingFace reference rows in the inference benchmarks
transformers>=4.51     # Qwen3 reference model + tokenizer (forward/tokenizer parity)
numpy                  # parity logit/token dumps
pillow                 # image I/O for the YOLO inference benchmark
ultralytics             # YOLOv8 reference row in tools/bench/bench_yolo_inference.py
safetensors             # every *_dump_reference.py fixture writer
diffusers               # tools/{vae,zimage,flux2,clip}_dump_reference.py, tools/vae_dump_reference.py
onnx                    # tools/arcface_dump_reference.py (ONNX SCRFD/ArcFace)
onnxruntime             # tools/arcface_dump_reference.py, tools/npu-adjacent NPU checks
opencv-python           # tools/arcface_dump_reference.py, tools/yolo_export/export_yolov8.py (cv2)
scikit-image            # tools/arcface_dump_reference.py (5-point alignment; skimage.transform)
pyyaml                  # tools/sam2_dump_reference.py, scripts/api_sync.py
soundfile                # scripts/brain_tts_client.py, tools/asr_dump_reference.py
sounddevice              # scripts/brain_tts_client.py, tools/asr_dump_reference.py (live audio)
jsonschema               # scripts/validate_spec.py (degrades to jq shape checks if absent)
chronos-forecasting      # tools/chronos2_dump_*_reference.py (`from chronos.chronos2 import ...`)
st_moe_pytorch           # tools/fincast_dump_reference.py
